// Isolated default-OFF experiment. Does not disable animation or physics.
class NFFollowerRootPolicy
{
    static bool Read;
    static bool Enabled;
    static bool DiagnosticsRequired;
    static bool IsEnabled()
    {
        if(!Read)
        {
            Read=true;
            if(GetGame().IsServer() && GetGame().IsMultiplayer())
            {
                Enabled=FileExist("$profile:nf-follower-root-skip.flag");
                DiagnosticsRequired=FileExist("$profile:nf-follower-root-diagnostics.flag");
                Print("NF_FOLLOWER_ROOT_POLICY enabled="+Enabled+" diagnostics="+DiagnosticsRequired);
            }
        }
        return Enabled && !DiagnosticsRequired;
    }
}
modded class Animal_DGNativeProbe
{
    // False means no current root sample: zero is never represented as measured.
    bool NFRootSampleCollected;
    int NFRootSampleSkips;
    int NFRootSampleQueries;
}
modded class DGNativeCommand
{
    override void RunPrePhys(float pDt)
    {
        if(!Retired && Owner && NFFollowerRootPolicy.IsEnabled())
        {
            if(Owner.GroundBody && !Owner.GroundLeader && !Owner.DamageReceiver && !Owner.ApplyRootMotion)
            {
                Owner.PhysicsUpdates++;
                Owner.NFRootSampleCollected=false;
                Owner.NFRootSampleSkips++;
                Owner.NativeRootRead=false;
                Owner.NativeRootDelta=vector.Zero;
                // NativeRootDistance retains historical sampled distance only.
                // Keep graph evaluation, command lifetime and zero translation.
                PrePhys_SetTranslation(vector.Zero);
                return;
            }
        }
        if(!Retired && Owner)
        {
            Owner.NFRootSampleCollected=true;
            Owner.NFRootSampleQueries++;
        }
        super.RunPrePhys(pDt);
    }
}
