        Director.Crowd=new DGBNativeCrowd(Director);Director.Pursuit=new DGBNativePursuit(Director);
        DGBNoise.Events.Clear();DGBNoise.Sequence=0;DGBNoise.Active=true;
        Print("NF_MATCHED_CUSTOM_READY logical="+count+" nativeParts="+(count*8));
    }
    void BeginRetirement()
    {
        if(!Director || Retiring){return;}
        Retiring=true;RetireAt=GetGame().GetTickTime();
        foreach(DGBNativeGroundBody body:Director.Bodies)
        {
            foreach(Animal_DGNativeProbe part:body.Parts){part.BeginRetirement();}
        }
        Print("NF_MATCHED_RETIRE_REQUESTED");
    }
    void FinishRetirement()
    {
        if(!Director || !Retiring){return;}
        int pending=0;int attached=0;
        foreach(DGBNativeGroundBody checked:Director.Bodies)
        {
            foreach(Animal_DGNativeProbe checkedPart:checked.Parts)
            {
                if(!checkedPart.RetirementObserved){pending++;}
                if(checkedPart.GetCommand_Script()){attached++;}
            }
        }
        float age=GetGame().GetTickTime()-RetireAt;
        if(age>8 && (pending>0 || attached>0)){Fail("retirement_timeout");return;}
        if(age<3 || pending>0 || attached>0){return;}
        ref array<Object> ownedEntities={};
        Director.SightExclusions.Clear();Director.PerceivedPlayers.Clear();Director.SightReports.Clear();Director.Player=null;
        foreach(DGBNativeGroundBody cleared:Director.Bodies)
        {
            foreach(Animal_DGNativeProbe oldPart:cleared.Parts){ownedEntities.Insert(oldPart);}
            cleared.Leader=null;cleared.Parts.Clear();cleared.Agent.Host=null;cleared.Agent.Filter=null;
            cleared.Agent.ConfirmedTarget=null;cleared.Agent.Suspect=null;cleared.Agent.SightExclusions.Clear();
            cleared.Agent.CollisionPlayers.Clear();cleared.Agent.BlockingPeer=null;cleared.Approach.ContactTarget=null;cleared.WaitingFor=null;
            if(cleared.Traversal){cleared.Traversal.Body=null;}
            cleared.Director=null;
        }
        Director.StaticMap.Records.Clear();Director.StaticMap.Director=null;
        Director.Traffic.Director=null;Director.Pursuit.Director=null;Director.Crowd.Director=null;
        Director.Bodies.Clear();Director.Agents.Clear();Director=null;
        foreach(Object removed:ownedEntities){GetGame().ObjectDelete(removed);}
        DGBNoise.Active=false;DGBNoise.Events.Clear();Retiring=false;
        if(Trace){CloseFile(Trace);Trace=null;}
        if(Costs){CloseFile(Costs);Costs=null;}
        LogsClosed=true;
        Print("NF_MATCHED_CUSTOM_RETIRED entities="+ownedEntities.Count());
    }
    void Update(float dt,PlayerBase player)
    {
        if(Failed || LogsClosed){return;}
        if(!Trace)
        {
            Trace=OpenFile("$profile:custom-agents.tsv",FileMode.WRITE);
            Costs=OpenFile("$profile:custom-costs.tsv",FileMode.WRITE);
            if(!Trace || !Costs){Fail("trace_files");return;}
            FPrintln(Trace,"time\tphase\tid\tcount\tposition\tworldPosition\tdistance\tvisible\ttargetPlayer\tstate\ttraversal\tnativeSeconds\tanimUpdates\trequest\tnativeRequest\tmoved\tparts");
            FPrintln(Costs,"time\tphase\tcount\tupdateTicks\tcommitTicks\tperceptionTicks\tsocialTicks\tpursuitTicks\tcrowdTicks\tstaticMapTicks\tdriveTicks\tcontactTicks\tboneTicks\tfootTicks\tworldTicks\tsolves\tfootRays\tworldSweeps\tpositionWrites\torientationWrites\tanimUpdates");
        }
        if(SeenRevision!=NFFlatStage.Revision)
        {
            SeenRevision=NFFlatStage.Revision;
            Print("NF_MATCHED_CLIENT_PHASE phase="+NFFlatStage.Phase+" count="+NFFlatStage.Count);
            if(NFFlatStage.Phase!=2 && NFFlatStage.Phase!=3){BeginRetirement();}
        }
        if(Retiring){FinishRetirement();return;}
        if(!Director && (NFFlatStage.Phase==2 || NFFlatStage.Phase==3))
        {
            Start(player,NFFlatStage.Count,NFFlatStage.Anchor);
        }
        if(!Director || Failed){return;}
        int updateAt=TickCount(0);Director.Update();
        int crowdAt=TickCount(0);Director.Crowd.Update(dt);Director.CrowdTicks+=TickCount(crowdAt);
        float now=GetGame().GetTickTime();
        foreach(DGBNativeGroundBody falling:Director.Bodies){falling.AdvanceFrameFall(now);}
        int commitAt=TickCount(0);
        foreach(DGBNativeGroundBody committed:Director.Bodies)
        {
            float targetDelay=0.08;
            if(committed.FrameFallPhysicsActive()){targetDelay=0.035;}
            committed.Presentation.SetDelayTarget(targetDelay,dt);
            float presentedHeading;
            vector expectedOrigin=committed.Presentation.Sample(committed.RenderOrigin(),committed.Agent.Heading,GetGame().GetTickTime(),presentedHeading);
            float expectedYaw=presentedHeading+180;
            foreach(Animal_DGNativeProbe rendered:committed.Parts)
            {
                if(vector.Distance(rendered.GetPosition(),expectedOrigin)>0.000001)
                {rendered.SetPosition(expectedOrigin);committed.PositionWrites++;}
                vector orientation=rendered.GetOrientation();float yawError=expectedYaw-orientation[0];
                while(yawError>180){yawError-=360;}
                while(yawError< -180){yawError+=360;}
                if(Math.AbsFloat(yawError)>0.0001 || Math.AbsFloat(orientation[1])>0.0001 || Math.AbsFloat(orientation[2])>0.0001)
                {rendered.SetOrientation(Vector(expectedYaw,0,0));committed.OrientationWrites++;}
            }
        }
        CommitTicks+=TickCount(commitAt);UpdateTicks+=TickCount(updateAt);
        if(now<NextSample){return;}
        NextSample=now+0.25;
        float drive,contact,bone,foot,world;
        int solves,rays,sweeps,writes,rotations,updates;
        foreach(DGBNativeGroundBody sampled:Director.Bodies)
        {
            drive+=sampled.DriveTicks;contact+=sampled.ContactTicks;bone+=sampled.ContactBoneTicks;
            foot+=sampled.ContactFootTicks;world+=sampled.ContactWorldTicks;solves+=sampled.ContactSolves;
            rays+=sampled.ContactFootRays;sweeps+=sampled.JointSweeps;writes+=sampled.PositionWrites;rotations+=sampled.OrientationWrites;
            foreach(Animal_DGNativeProbe counted:sampled.Parts){updates+=counted.AnimUpdates;}
            bool target=sampled.Agent.ConfirmedTarget==player;
            string row=now.ToString()+"\t"+NFFlatStage.Phase+"\t"+sampled.Id+"\t"+Director.Bodies.Count();
            row+="\t"+sampled.Agent.Position.ToString(false)+"\t"+sampled.Leader.GetPosition().ToString(false);
            row+="\t"+vector.Distance(sampled.Agent.Position,player.GetPosition())+"\t"+sampled.Agent.Visible+"\t"+target;
            row+="\t"+sampled.Agent.State+"\t"+sampled.Traversal.Phase+"\t"+sampled.Leader.NativeSeconds;
            row+="\t"+sampled.Leader.AnimUpdates+"\t"+sampled.Request+"\t"+sampled.Leader.NativeRequest;
            row+="\t"+sampled.Agent.Moved+"\t"+sampled.Parts.Count();FPrintln(Trace,row);
        }
        string costRow=now.ToString()+"\t"+NFFlatStage.Phase+"\t"+Director.Bodies.Count();
        costRow+="\t"+UpdateTicks+"\t"+CommitTicks+"\t"+Director.PerceptionTicks+"\t"+Director.SocialTicks;
        costRow+="\t"+Director.PursuitTicks+"\t"+Director.CrowdTicks+"\t"+Director.StaticMap.CpuTicks;
        costRow+="\t"+drive+"\t"+contact+"\t"+bone+"\t"+foot+"\t"+world;
        costRow+="\t"+solves+"\t"+rays+"\t"+sweeps+"\t"+writes+"\t"+rotations+"\t"+updates;
        FPrintln(Costs,costRow);
    }
}

        super.OnUpdate(timeslice);

        NFFlat.Update(timeslice,player);
    }
