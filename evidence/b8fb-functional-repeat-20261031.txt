    static bool ArrivalAllowed(bool held,float gap,bool sameGoal)
    {
        // Enter at the original precise endpoint. Once physically reached,
        // bounded crowd pressure may move the actor within this terminal band
        // without repeatedly restarting locomotion. No goal/root is rewritten.
        return gap<0.18 || (held && sameGoal && gap<0.45);
    }
    float ArrivalRadius(DGBNativeGroundBody body)
    {
        DGBNativeApproachState state=body.Approach;
        float gap=DGBGroundAgent.FlatDistance(body.Agent.Position,state.Goal);
        bool same=vector.Distance(state.ArrivalGoal,state.Goal)<0.05;
        bool held=state.Arrived && same;
        bool arrived=ArrivalAllowed(held,gap,same) && Math.AbsFloat(body.Agent.Position[1] - state.Goal[1])<0.2;
        if(arrived && !held)
        {state.ArrivalEntries++;state.ArrivalGoal=state.Goal;state.ArrivalPosition=body.Agent.Position;state.ArrivalEntryGap=gap;}
        state.Arrived=arrived;
        if(arrived){state.ArrivalHolds++;return 0.45;}
        return 0.18;
    }
