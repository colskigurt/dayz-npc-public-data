import numpy as np
ROI=(730,300,150,115)
CAP_KERNELS=[]
for radius in [9,10,11,12,13]:
 yy,xx=np.mgrid[:radius+6,:2*radius+7];cx=radius+3;cy=radius+3
 inside=(xx-cx)**2+(yy-cy)**2<=radius*radius
 kernel=inside/inside.sum()-(~inside)/(~inside).sum()
 CAP_KERNELS.append((radius,kernel.shape,np.fft.rfft2(kernel[::-1,::-1],s=(256,256))))
def track(rgb,threshold=145):
 # Image-only circular upper-head silhouette contrast. No CPU/HUD/previous-point input.
 a=rgb.astype(np.int16);r,g,b=a[:,:,0],a[:,:,1],a[:,:,2]
 mask=(r>=threshold)&(g>=threshold+5)&(b>=threshold-25)&(g>=r)&((g-b)>=10)&((g-b)<=70)
 spectrum=np.fft.rfft2(mask.astype(float),s=(256,256));best=None
 for radius,(kh,kw),kernel in CAP_KERNELS:
  conv=np.fft.irfft2(spectrum*kernel,s=(256,256))[kh-1:mask.shape[0],kw-1:mask.shape[1]]
  idx=np.unravel_index(np.argmax(conv),conv.shape);score=float(conv[idx])
  row=dict(x=float(idx[1]+radius+3)+ROI[0],y=float(idx[0]+3)+ROI[1],radius=radius,score=score)
  if best is None or score>best['score']:best=row
 return best
