    bool GroundScopeEligible(vector anchor)
    {
        if(!Agent || !Traversal)return false;
        if(GroundPlatform || Traversal.Phase!="FOLLOW" || Traversal.Area!="ground")return false;
        if(Agent.State=="DEAD")return false;
        if(Combat && (Combat.Active || Combat.Dead || Combat.PendingDeath))return false;
        int count=Agent.ObstacleMin.Count();
        if(count!=Agent.ObstacleMax.Count())return false;
        for(int i=0;i<count;i++)
        {
            vector low=Agent.ObstacleMin[i];vector high=Agent.ObstacleMax[i];
            if(anchor[0]<low[0] - 1.5 || anchor[0]>high[0]+1.5)continue;
            if(anchor[2]<low[2] - 1.5 || anchor[2]>high[2]+1.5)continue;
            return false;
        }
        return true;
    }
