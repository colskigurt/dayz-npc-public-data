def number(v):

def distribution(v):

 a=sorted(v)
 def q(f):
  x=(len(a)-1)*f;i=int(x);return a[i]+(a[min(i+1,len(a)-1)]-a[i])*(x-i)
 return dict(samples=len(v),p50=q(.5),p95=q(.95),p99=q(.99),minimum=a[0],maximum=a[-1])
