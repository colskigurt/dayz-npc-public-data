Exact periodic basis and cycle-weighted fit functions:
def basis(t,k,p=period):
    phase=np.asarray(t,dtype=float)*2*np.pi/p
    return np.column_stack([np.ones(len(phase))]+[f(j*phase) for j in range(1,k+1) for f in [np.sin,np.cos]])

def fit(t,y,k,groups,p=period):
    x=basis(t,k,p);unique,counts=np.unique(groups,return_counts=True);weights=dict(zip(unique,counts))
    w=np.sqrt(np.asarray([1/weights[g] for g in groups]))
    beta,_,rank,_=np.linalg.lstsq(x*w[:,None],y*w[:,None],rcond=None)
    assert rank==x.shape[1]
    return beta
