        DGFollowCost.Record(this,7,TickCount(started));
    }
    override void AdvanceFrameFall(float now)
    {
        super.AdvanceFrameFall(now);
        if(DGFollowCost.On() && Director && Id==Director.Bodies.Count() - 1)DGFollowCost.Sample(Director);
    }
